155 void
156 initialize(const std::string &file_path)
157 {
158 std::lock_guard<std::mutex> lock(lock_);
159 if (!initialized_)
160 {
161 conf_ = loadConf(file_path);
162 validateConf(conf_);
163 initialized_ = true;
164 }
165 }
166
167 const ConfHierBEM &
168 getConfig() const
169 {
170 if (!initialized_)
171 {
172 throw std::runtime_error("ConfigFile not initialized");
173 }
174 return conf_;
175 }
176
177private:
178 ConfigFile()
179 : initialized_(false)
180 {}
181 ConfigFile(const ConfigFile &) = delete;
182 ConfigFile &
183 operator=(const ConfigFile &) = delete;
184
185 ConfHierBEM
186 loadConf(const std::string &file_path)
187 {
188 auto &conf =
189 rfl::toml::load<ConfHierBEM, rfl::DefaultIfMissing, rfl::NoExtraFields>(
190 file_path)
191 .value();
192 return conf;
193 }
194
195 void
196 validateConf(const ConfHierBEM &conf)
197 {
198 if (conf.bem.space_dim.value() != 3 || conf.bem.boundary_dim.value() != 2)
199 {
200 throw std::runtime_error("Currently only 3D/2D problems are supported");
201 }
202
203 if (conf.bem.space_dim.value() <= conf.bem.boundary_dim.value())
204 {
205 throw std::runtime_error(
206 "'space_dim' must be greater than 'boundary_dim'");
207 }
208 }
209
210 ConfHierBEM conf_;
211 bool initialized_;
212 std::mutex lock_;
213};
