258template <int dim,
259 int spacedim,
260 typename SearchableMaterialIdContainer,
261 template <int, typename>
262 typename KernelFunctionType,
263 typename RangeNumberType,
264 typename KernelNumberType>
265void
266BEMBilinearForm<dim,
267 spacedim,
268 SearchableMaterialIdContainer,
269 KernelFunctionType,
270 RangeNumberType,
271 KernelNumberType>::build_block_cluster_tree(const double eta,
272 const unsigned int
273 n_min)
274{
275 // When building a block cluster tree, the test space appears before the trial
276 // space, since the test space is related to matrix rows, while the trial
277 // space is related to matrix columns.
278 block_cluster_tree = std::make_unique<BlockClusterTree<spacedim, real_type>>(
279 test_space.get_cluster_tree(), trial_space.get_cluster_tree(), eta, n_min);
280
281 if (&trial_space == &test_space)
282 block_cluster_tree->partition(
283 trial_space.get_internal_to_external_dof_numbering(),
284 trial_space.get_support_points(),
285 trial_space.get_dof_average_cell_size());
286 else
287 block_cluster_tree->partition(
288 test_space.get_internal_to_external_dof_numbering(),
289 trial_space.get_internal_to_external_dof_numbering(),
290 test_space.get_support_points(),
291 trial_space.get_support_points(),
292 test_space.get_dof_average_cell_size(),
293 trial_space.get_dof_average_cell_size());
294}

297template <int dim,
298 int spacedim,
299 typename SearchableMaterialIdContainer,
300 template <int, typename>
301 typename KernelFunctionType,
302 typename RangeNumberType,
303 typename KernelNumberType>
304std::unique_ptr<HMatrix<spacedim, RangeNumberType>>
305BEMBilinearForm<dim,
306 spacedim,
307 SearchableMaterialIdContainer,
308 KernelFunctionType,
309 RangeNumberType,
310 KernelNumberType>::
311 build_hmatrix(const unsigned int thread_num,
312 const unsigned int max_rank,
313 const real_type epsilon,
314 const DeviceNumberType<KernelNumberType> kernel_factor,
315 const SauterQuadratureRule<dim> &sauter_quad_rule,
316 const std::vector<MappingInfo<dim, spacedim> *> &mappings,
317 const std::map<types::material_id, unsigned int>
318 &material_id_to_mapping_index,
319 SubdomainTopology<dim, spacedim> &subdomain_topology)
320{
321 // The kernel does not need regularization.
322 Assert(!kernel.needs_regularization(), ExcInternalError());
323
324 HMatrixSupport::Property property = is_symmetric ?
325 HMatrixSupport::Property::symmetric :
326 HMatrixSupport::Property::general;
327 HMatrixSupport::BlockType block_type =
328 HMatrixSupport::BlockType::diagonal_block;
329 auto hmat = std::make_unique<HMatrix<spacedim, RangeNumberType>>(
330 *block_cluster_tree, max_rank, property, block_type);
331
332 fill_hmatrix_with_aca_plus_smp<dim,
333 spacedim,
334 KernelFunctionType,
335 RangeNumberType,
336 KernelNumberType,
337 SurfaceNormalDetector<dim, spacedim>>(
338 thread_num,
339 *hmat,
340 ACAConfig<real_type>(max_rank, epsilon, block_cluster_tree->get_eta()),
341 kernel,
342 kernel_factor,
343 test_space.get_dof_to_cell_topo(),
344 trial_space.get_dof_to_cell_topo(),
345 sauter_quad_rule,
346 test_space.get_dof_handler(),
347 trial_space.get_dof_handler(),
348 test_space.get_is_full_domain() ?
349 nullptr :
350 &test_space.get_local_to_full_dof_id_map(),
351 trial_space.get_is_full_domain() ?
352 nullptr :
353 &trial_space.get_local_to_full_dof_id_map(),
354 test_space.get_internal_to_external_dof_numbering(),
355 trial_space.get_internal_to_external_dof_numbering(),
356 mappings,
357 material_id_to_mapping_index,
358 SurfaceNormalDetector<dim, spacedim>(subdomain_topology),
359 is_symmetric);
360
361 return hmat;
362}
