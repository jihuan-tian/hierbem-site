41 const size_t stack_size = 1024 * 10;
42 AssertCuda(cudaDeviceSetLimit(cudaLimitStackSize, stack_size));
43 AssertCuda(
44 cudaGetDeviceProperties(&HierBEM::CUDAWrappers::device_properties, 0));
45
46 const unsigned int dim = 2;
47 const unsigned int spacedim = 3;
48
49 LaplaceBEM<dim, spacedim, double, double> bem(
50 1, // fe order for dirichlet space
51 0, // fe order for neumann space
52 ProblemType::MixedBCProblem,
53 true, // is interior problem
54 32, // n_min for cluster tree
55 32, // n_min for block cluster tree
56 0.8, // eta for H-matrix
57 10, // max rank for H-matrix
58 0.01, // aca epsilon for H-matrix
59 1.0, // eta for preconditioner
60 5, // max rank for preconditioner
61 0.1, // aca epsilon for preconditioner
62 MultithreadInfo::n_threads() // Number of threads used for ACA
63 );
64 bem.set_project_name("solve-laplace-mixed");
65 bem.set_preconditioner_type(PreconditionerType::OperatorPreconditioning);
66 bem.set_iterative_solver_vmult_type(
67 IterativeSolverVmultType::SerialIterative);
68
69 std::ifstream mesh_in(HBEM_TEST_MODEL_DIR "cylinder.msh");
70 read_msh(mesh_in, bem.get_triangulation());
71 bem.get_subdomain_topology().generate_topology(HBEM_TEST_MODEL_DIR
72 "cylinder.brep",
73 HBEM_TEST_MODEL_DIR
74 "cylinder.msh");
75 bem.get_triangulation().refine_global(1);
76
77 // Create the map from manifold ids to manifold objects. Because in the
78 // destructor of LaplaceBEM the manifold objects will be released, the
79 // manifold object here is created on the heap.
80 FlatManifold<dim, spacedim> *flat_manifold =
81 new FlatManifold<dim, spacedim>();
82 CylindricalManifold<dim, spacedim> *cyl_manifold =
83 new CylindricalManifold<dim, spacedim>(2);
84 bem.get_manifolds()[0] = flat_manifold;
85 bem.get_manifolds()[1] = cyl_manifold;
86
87 // Create the map from material ids to manifold ids.
88 bem.get_manifold_description()[1] = 1;
89 bem.get_manifold_description()[2] = 0;
90 bem.get_manifold_description()[3] = 0;
91
92 // Create the map from manifold id to mapping order.
93 bem.get_manifold_id_to_mapping_order()[0] = 1;
94 bem.get_manifold_id_to_mapping_order()[1] = 2;
95
96 Functions::ConstantFunction<spacedim> f1(1.0);
97 Functions::ConstantFunction<spacedim> f2(0.0);
98 bem.assign_neumann_bc(f2, 1);
99 bem.assign_dirichlet_bc(f1, 2);
100 bem.assign_dirichlet_bc(f2, 3);
101
102 bem.run();
103}
