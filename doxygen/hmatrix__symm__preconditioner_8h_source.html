358template <int spacedim, typename Number>
359void
360HMatrixSymmPreconditioner<spacedim, Number>::vmult(
361 Vector<Number> &y,
362 const Vector<Number> &x) const
363{
364 if (is_spd)
365 this->solve_cholesky(y, x);
366 else
367 this->solve_lu(y, x);
368}
