47// Builder class for cluster tree.
48template <int spacedim>
49class ClusterTreeBuilder
50{
51public:
52 template <int dim>
53 ClusterTreeBuilder(const Mapping<dim, spacedim> &mapping,
54 const DoFHandler<dim, spacedim> &dof_handler,
55 const unsigned int _n_min);
56
57 // Build a cluster tree and return it as a unique smart pointer, since it will
58 // be associated with a unique function space.
59 std::unique_ptr<ClusterTree<spacedim>>
60 build() const;
61
62 std::vector<Point<spacedim>> &
63 get_support_points()
64 {
65 return support_points;
66 }
67
68 const std::vector<Point<spacedim>> &
69 get_support_points() const
70 {
71 return support_points;
72 }
73
74 std::vector<double> &
75 get_dof_average_cell_size()
76 {
77 return dof_average_cell_size;
78 }
79
80 const std::vector<double> &
81 get_dof_average_cell_size() const
82 {
83 return dof_average_cell_size;
84 }
85
86private:
87 // List of support points in the finite element associated with this cluster
88 // tree.
89 std::vector<Point<spacedim>> support_points;
90 // List of DoF indices in the finite element, starting from zero.
91 std::vector<types::global_dof_index> dof_indices;
92 // List of average cell size estimated at support points.
93 std::vector<double> dof_average_cell_size;
94 // Minimum number of DoFs in a cluster. When the actual number of DoFs in a
95 // cluster is smaller than @p n_min , the cluster is a leaf node in the
96 // cluster tree.
97 unsigned int n_min;
98};

101template <int spacedim>
102template <int dim>
103ClusterTreeBuilder<spacedim>::ClusterTreeBuilder(
104 const Mapping<dim, spacedim> &mapping,
105 const DoFHandler<dim, spacedim> &dof_handler,
106 const unsigned int _n_min)
107 : n_min(_n_min)
108{
109 Assert(dof_handler.get_fe().has_support_points(), ExcInternalError());
110
111 const types::global_dof_index n_dofs = dof_handler.n_dofs();
112
113 // Get the coordinates for all support points.
114 support_points.resize(n_dofs);
115 DoFTools::map_dofs_to_support_points(mapping, dof_handler, support_points);
116
117 // Generate a list of DoF indices starting from zero.
118 dof_indices.resize(n_dofs);
119 for (types::global_dof_index d = 0; d < n_dofs; d++)
120 dof_indices[d] = d;
121
122 // Calculate the average mesh cell size at each support point.
123 dof_average_cell_size.assign(n_dofs, 0);
124 DoFToolsExt::map_dofs_to_average_cell_size(dof_handler,
125 dof_average_cell_size);
126}
127
128
129template <int spacedim>
130std::unique_ptr<ClusterTree<spacedim>>
131ClusterTreeBuilder<spacedim>::build() const
132{
133 // Create a cluster tree for all the DoF indices.
134 auto cluster_tree = std::make_unique<ClusterTree<spacedim>>(
135 dof_indices, support_points, dof_average_cell_size, n_min);
136 // Partition the cluster tree.
137 cluster_tree->partition(support_points, dof_average_cell_size);
138
139 return cluster_tree;
140}

315template <int dim, int spacedim>
316void
317BEMBilinearForm<dim, spacedim>::build_block_cluster_tree(
318 const double eta,
319 const unsigned int n_min)
320{
321 // When building a block cluster tree, the test space appears before the trial
322 // space, since the test space is related to matrix rows, while the trial
323 // space is related to matrix columns.
324 block_cluster_tree = std::make_unique<BlockClusterTree<spacedim>>(
325 test_space.get_cluster_tree(), trial_space.get_cluster_tree(), eta, n_min);
326 block_cluster_tree->partition(
327 test_space.get_internal_to_external_dof_numbering(),
328 trial_space.get_internal_to_external_dof_numbering(),
329 test_space.get_support_points(),
330 trial_space.get_support_points(),
331 test_space.get_dof_average_cell_size(),
332 trial_space.get_dof_average_cell_size());
333}

336int
337main()
338{
339 // Generate a triangulation for a unit sphere.
340 const unsigned int dim = 2;
341 const unsigned int spacedim = 3;
342 const Point<spacedim> center(0, 0, 0);
343 const double radius(1);
344 const unsigned int refinement = 1;
345
346 Triangulation<dim, spacedim> tria;
347 GridGenerator::hyper_sphere(tria, center, radius);
348 tria.refine_global(refinement);
349
350 // Create a mapping object for transforming unit support points in the unit
351 // cell to all real cells.
352 const unsigned int mapping_order = 2;
353 const MappingQ<dim, spacedim> mapping(mapping_order);
354
355 // Parameters for cluster trees and block cluster tree.
356 const unsigned int n_min_H_half = 4;
357 const unsigned int n_min_H_minus_half = 4;
358 const double eta = 4;
359 const unsigned int n_min_block_cluster_tree = 4;
360
361 // Create a continuous Lagrangian finite element and a DoF handler for the
362 // Sobolev space \f$H^{1/2}(\Gamma)\f$.
363 const unsigned int fe_H_half_order = 1;
364 FE_Q<dim, spacedim> fe_H_half(fe_H_half_order);
365 DoFHandler<dim, spacedim> dof_handler_H_half(tria);
366 dof_handler_H_half.distribute_dofs(fe_H_half);
367 BEMFunctionSpace<dim, spacedim> H_half(dof_handler_H_half,
368 mapping,
369 n_min_H_half);
370
371 // Create a discontinuous Lagrangian finite element and a DoF handler for the
372 // Sobolev space \f$H^{-1/2}(\Gamma)\f$ space.
373 const unsigned int fe_H_minus_half_order = 0;
374 FE_DGQ<dim, spacedim> fe_H_minus_half(fe_H_minus_half_order);
375 DoFHandler<dim, spacedim> dof_handler_H_minus_half(tria);
376 dof_handler_H_minus_half.distribute_dofs(fe_H_minus_half);
377 BEMFunctionSpace<dim, spacedim> H_minus_half(dof_handler_H_minus_half,
378 mapping,
379 n_min_H_minus_half);
380
381 // Create a bilinear form \f$b_V: H^{-1/2}(\Gamma)\times H^{-1/2}(\Gamma)
382 // \rightarrow \mathbb{R}\f$ for the single layer potential operator \f$V\f$.
383 BEMBilinearForm<dim, spacedim> bV(H_minus_half, H_minus_half);
384 bV.build_block_cluster_tree(eta, n_min_block_cluster_tree);
385 // Create a bilinear form \f$b_K: H^{1/2}(\Gamma)\times H^{-1/2}(\Gamma)
386 // \rightarrow \mathbb{R}\f$ for the double layer potential operator \f$K\f$.
387 BEMBilinearForm<dim, spacedim> bK(H_half, H_minus_half);
388 bK.build_block_cluster_tree(eta, n_min_block_cluster_tree);
389
390 // Print out the cluster trees for the two function spaces.
391 std::ofstream graph("cluster-tree-H-half.puml");
392 H_half.get_cluster_tree().print_tree_info_as_dot(graph);
393 graph.close();
394
395 graph.open("cluster-tree-H-minus-half.puml");
396 H_minus_half.get_cluster_tree().print_tree_info_as_dot(graph);
397 graph.close();
398
399 // Print out the block cluster trees for the two bilinear forms.
400 graph.open("bV-block-cluster-tree.puml");
401 bV.get_block_cluster_tree().print_bct_info_as_dot(graph);
402 graph.close();
403
404 graph.open("bK-block-cluster-tree.puml");
405 bK.get_block_cluster_tree().print_bct_info_as_dot(graph);
406 graph.close();
407}
