145template <int dim,
146 int spacedim,
147 typename RangeNumberType,
148 typename KernelNumberType>
149HelmholtzAcousticHyperSingularPreconditioner<dim,
150 spacedim,
151 RangeNumberType,
152 KernelNumberType>::
153 HelmholtzAcousticHyperSingularPreconditioner(
154 FiniteElement<dim, spacedim> &fe_primal_space,
155 FiniteElement<dim, spacedim> &fe_dual_space,
156 const Triangulation<dim, spacedim> &tria,
157 const std::vector<types::global_dof_index> &primal_space_dof_i2e_numbering,
158 const std::vector<types::global_dof_index> &primal_space_dof_e2i_numbering,
159 const unsigned int max_iter,
160 const real_type tol,
161 const real_type omega,
162 const bool log_history,
163 const bool log_result)
164 : HyperSingularPreconditioner<dim, spacedim, RangeNumberType>(
165 fe_primal_space,
166 fe_dual_space,
167 tria,
168 primal_space_dof_i2e_numbering,
169 primal_space_dof_e2i_numbering,
170 max_iter,
171 tol,
172 omega,
173 log_history,
174 log_result)
175{
176 // At the moment, in a Neumann problem, the primal space can only be
177 // @p FE_Q(1) and the dual space can only be @p FE_DGQ(0). Therefore, we make
178 // assertions here about their numbers of DoFs in a cell and the conformity
179 // with continuous function spaces.
180 AssertDimension(this->fe_primal_space.dofs_per_cell, 4);
181 Assert(this->fe_primal_space.conforms(FiniteElementData<dim>::Conformity::H1),
182 ExcInternalError());
183 AssertDimension(this->fe_dual_space.dofs_per_cell, 1);
184 Assert(this->fe_dual_space.conforms(FiniteElementData<dim>::Conformity::L2),
185 ExcInternalError());
186}
