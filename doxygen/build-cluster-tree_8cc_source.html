43main()
44{
45 // Generate a triangulation for a unit sphere.
46 const unsigned int dim = 2;
47 const unsigned int spacedim = 3;
48 const Point<spacedim> center(0, 0, 0);
49 const double radius(1);
50 const unsigned int refinement = 2;
51
52 Triangulation<dim, spacedim> tria;
53 GridGenerator::hyper_sphere(tria, center, radius);
54 tria.refine_global(refinement);
55
56 // Save the mesh to a file for visualization.
57 GridOut grid_out;
58 std::ofstream mesh_file("sphere.msh");
59 grid_out.write_msh(tria, mesh_file);
60
61 // Create a continuous Lagrangian finite element and a DoF handler.
62 const unsigned int fe_order = 1;
63 FE_Q<dim, spacedim> fe(fe_order);
64 DoFHandler<dim, spacedim> dof_handler(tria);
65 dof_handler.distribute_dofs(fe);
66
67 // Create a mapping object for transforming unit support points in the unit
68 // cell to all real cells.
69 const unsigned int mapping_order = 2;
70 const MappingQ<dim, spacedim> mapping(mapping_order);
71 // Get the coordinates for all support points.
72 std::vector<Point<spacedim>> support_points(dof_handler.n_dofs());
73 DoFTools::map_dofs_to_support_points(mapping, dof_handler, support_points);
74
75 // Generate a list of DoF indices starting from zero.
76 std::vector<types::global_dof_index> dof_indices(dof_handler.n_dofs());
77 for (types::global_dof_index d = 0; d < dof_indices.size(); d++)
78 dof_indices[d] = d;
79
80 // Create a cluster tree for all the DoF indices.
81 const unsigned int n_min = 4;
82 ClusterTree<spacedim> cluster_tree(dof_indices, support_points, n_min);
83 // Partition the cluster tree.
84 cluster_tree.partition(support_points);
85
86 // Export the cluster tree as a directed graph.
87 std::ofstream graph("cluster-tree.puml");
88 cluster_tree.print_tree_info_as_dot(graph);
89 graph.close();
90}
