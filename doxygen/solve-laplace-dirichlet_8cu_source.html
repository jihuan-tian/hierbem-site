45// Function object for the Dirichlet boundary condition data.
46class DirichletBC : public Function<3>
47{
48public:
49 DirichletBC()
50 : Function<3>()
51 , x0(0., 0., 0.)
52 {}
53
54 DirichletBC(const Point<3> &x0_)
55 : Function<3>()
56 , x0(x0_)
57 {}
58
59 double
60 value(const Point<3> &p, const unsigned int component = 0) const
61 {
62 (void)component;
63 return 1.0 / 4.0 / numbers::PI / (p - x0).norm();
64 }
65
66private:
67 // Location of the unit Dirac source.
68 Point<3> x0;
69};

72// Function object for the analytical solution.
73class AnalyticalSolution : public Function<3>
74{
75public:
76 AnalyticalSolution()
77 : Function<3>()
78 , x0(0., 0., 0.)
79 , center(0.0, 0.0, 0.0)
80 , radius(1.0)
81 {}
82
83 AnalyticalSolution(const Point<3> &x0_,
84 const Point<3> &center_,
85 double radius_)
86 : Function<3>()
87 , x0(x0_)
88 , center(center_)
89 , radius(radius_)
90 {}
91
92 double
93 value(const Point<3> &p, const unsigned int component = 0) const
94 {
95 (void)component;
96
97 Tensor<1, 3> diff_vector = x0 - p;
98
99 return ((p - center) * diff_vector) / 4.0 / numbers::PI /
100 std::pow(diff_vector.norm(), 3) / radius;
101 }
102
103private:
104 // Location of the unit Dirac source.
105 Point<3> x0;
106 Point<3> center;
107 double radius;
108};

110template <int dim,
111 int spacedim,
112 typename RangeNumberType,
113 typename KernelNumberType>
114void
115output_analytical_solution(
116 const LaplaceBEM<dim, spacedim, RangeNumberType, KernelNumberType> &bem,
117 const Point<spacedim> &source_loc,
118 const Point<spacedim> &center,
119 const double radius)
120{
121 const DoFHandler<dim, spacedim> &dof_handler = bem.get_dof_handler_neumann();
122 Vector<double> analytical_solution(dof_handler.n_dofs());
123 // We use the 2nd order mapping to interpolate the analytical solution.
124 VectorTools::interpolate(bem.get_mappings()[1]->get_mapping(),
125 dof_handler,
126 AnalyticalSolution(source_loc, center, radius),
127 analytical_solution);
128 std::ofstream out("analytical-solution.output");
129 print_vector_to_mat(out, "analytical_solution", analytical_solution);
130 out.close();
131}

141int
142main()
143{
144 const size_t stack_size = 1024 * 10;
145 AssertCuda(cudaDeviceSetLimit(cudaLimitStackSize, stack_size));
146 AssertCuda(
147 cudaGetDeviceProperties(&HierBEM::CUDAWrappers::device_properties, 0));
148
149 const unsigned int dim = 2;
150 const unsigned int spacedim = 3;
151
152 LaplaceBEM<dim, spacedim, double, double> bem(
153 1, // fe order for dirichlet space
154 0, // fe order for neumann space
155 ProblemType::DirichletBCProblem,
156 true, // is interior problem
157 32, // n_min for cluster tree
158 32, // n_min for block cluster tree
159 0.8, // eta for H-matrix
160 10, // max rank for H-matrix
161 0.01, // aca epsilon for H-matrix
162 1.0, // eta for preconditioner
163 5, // max rank for preconditioner
164 0.1, // aca epsilon for preconditioner
165 MultithreadInfo::n_threads() // Number of threads used for ACA
166 );
167 bem.set_project_name("solve-laplace-dirichlet");
168 bem.set_preconditioner_type(PreconditionerType::OperatorPreconditioning);
169 bem.set_iterative_solver_vmult_type(
170 IterativeSolverVmultType::SerialIterative);
171
172 const Point<spacedim> center(0, 0, 0);
173 const double radius(1);
174 const unsigned int refinement = 4;
175
176 Triangulation<dim, spacedim> tria;
177 GridGenerator::hyper_sphere(tria, center, radius);
178 tria.refine_global(refinement);
179 bem.get_triangulation() = std::move(tria);
180
181 // Create the map from material ids to manifold ids. By default, the material
182 // ids of all cells are zero, if the triangulation is created by a deal.ii
183 // function in GridGenerator.
184 bem.get_manifold_description()[0] = 0;
185
186 // Create the map from manifold ids to manifold objects. Because in the
187 // destructor of LaplaceBEM the manifold objects will be released, the
188 // manifold object here is created on the heap.
189 SphericalManifold<dim, spacedim> *spherical_manifold =
190 new SphericalManifold<dim, spacedim>(center);
191 bem.get_manifolds()[0] = spherical_manifold;
192
193 // Create the map from manifold id to mapping order.
194 bem.get_manifold_id_to_mapping_order()[0] = 2;
195
196 // Build surface-to-volume and volume-to-surface relationship.
197 bem.get_subdomain_topology().generate_single_domain_topology_for_dealii_model(
198 {0});
199
200 Point<spacedim> source_loc(1., 1., 1.);
201 DirichletBC dirichlet_bc(source_loc);
202 bem.assign_dirichlet_bc(dirichlet_bc);
203
204 bem.run();
205
206 output_analytical_solution(bem, source_loc, center, radius);
207}
