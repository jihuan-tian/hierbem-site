174template <int dim,
175 int spacedim,
176 typename RangeNumberType,
177 typename KernelNumberType>
178HelmholtzAcousticSingleLayerPreconditioner<dim,
179 spacedim,
180 RangeNumberType,
181 KernelNumberType>::
182 HelmholtzAcousticSingleLayerPreconditioner(
183 FiniteElement<dim, spacedim> &fe_primal_space,
184 FiniteElement<dim, spacedim> &fe_dual_space,
185 const Triangulation<dim, spacedim> &tria,
186 const std::vector<types::global_dof_index> &primal_space_dof_i2e_numbering,
187 const std::vector<types::global_dof_index> &primal_space_dof_e2i_numbering,
188 const unsigned int max_iter,
189 const real_type tol,
190 const real_type omega,
191 const bool log_history,
192 const bool log_result)
193 : SingleLayerPreconditioner<dim, spacedim, RangeNumberType>(
194 fe_primal_space,
195 fe_dual_space,
196 tria,
197 primal_space_dof_i2e_numbering,
198 primal_space_dof_e2i_numbering,
199 max_iter,
200 tol,
201 omega,
202 log_history,
203 log_result)
204{
205 // At the moment, in a Dirichlet problem, the primal space can only be
206 // @p FE_DGQ(0) and the dual space can only be @p FE_Q(1). Therefore, we make
207 // assertions here about their numbers of DoFs in a cell and the conformity
208 // with continuous function spaces.
209 AssertDimension(this->fe_primal_space.dofs_per_cell, 1);
210 Assert(this->fe_primal_space.conforms(FiniteElementData<dim>::Conformity::L2),
211 ExcInternalError());
212 AssertDimension(this->fe_dual_space.dofs_per_cell, 4);
213 Assert(this->fe_dual_space.conforms(FiniteElementData<dim>::Conformity::H1),
214 ExcInternalError());
215}

264template <int dim,
265 int spacedim,
266 typename RangeNumberType,
267 typename KernelNumberType>
268template <typename SurfaceNormalDetector>
269void
270HelmholtzAcousticSingleLayerPreconditioner<dim,
271 spacedim,
272 RangeNumberType,
273 KernelNumberType>::
274 setup_preconditioner(
275 const unsigned int thread_num,
276 const HMatrixParameters<real_type> &hmat_params,
277 const SubdomainTopology<dim, spacedim> &subdomain_topology,
278 const std::vector<MappingInfo<dim, spacedim> *> &mappings,
279 const std::map<types::material_id, unsigned int>
280 &material_id_to_mapping_index,
281 const SurfaceNormalDetector &normal_detector,
282 const SauterQuadratureRule<dim> &sauter_quad_rule,
283 const Quadrature<dim> &quad_rule_for_mass)
284{
285 // Call the parent class's function to setup the preconditioner, but without
286 // building the H-matrix on the refined mesh.
287 OperatorPreconditioner<dim, spacedim, RangeNumberType>::setup_preconditioner(
288 hmat_params, mappings, quad_rule_for_mass);
289
290 // Build the H-matrix for the first part of the regularized bilinear form for
291 // the hyper singular boundary integral operator.
292 this->template build_preconditioner_hmat_on_refined_mesh<
293 PlatformShared::HelmholtzAcousticKernel::HyperSingularKernelRegular1,
294 KernelNumberType>(this->preconditioner_hmat,
295 hmat_params,
296 hyper_singular_kernel1,
297 thread_num,
298 subdomain_topology,
299 mappings,
300 material_id_to_mapping_index,
301 normal_detector,
302 sauter_quad_rule);
303
304 // Build the H-matrix for the second part of the regularized bilinear form for
305 // the hyper singular boundary integral operator.
306 this->template build_preconditioner_hmat_on_refined_mesh<
307 PlatformShared::HelmholtzAcousticKernel::HyperSingularKernelRegular2,
308 KernelNumberType>(preconditioner_hmat2,
309 hmat_params,
310 hyper_singular_kernel2,
311 thread_num,
312 subdomain_topology,
313 mappings,
314 material_id_to_mapping_index,
315 normal_detector,
316 sauter_quad_rule);
317
318 // Add the second part H-matrix into the first part. N.B. The truncation rank
319 // for the formatted addition is set to 2*max_hmat_rank, so the formatted
320 // addition is accurate.
321 this->preconditioner_hmat.add(preconditioner_hmat2,
322 hmat_params.max_hmat_rank * 2);
323 preconditioner_hmat2.release();
324}
