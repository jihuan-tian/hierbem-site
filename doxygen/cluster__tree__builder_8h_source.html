135template <int spacedim, typename Number>
136template <int dim>
137ClusterTreeBuilder<spacedim, Number>::ClusterTreeBuilder(
138 const DoFHandler<dim, spacedim> &dof_handler,
139 const unsigned int n_min_)
140 : n_min(n_min_)
141{
142 Assert(dof_handler.get_fe().has_support_points(), ExcInternalError());
143
144 const types::global_dof_index n_dofs = dof_handler.n_dofs();
145
146 // Get the coordinates for all support points.
147 support_points.resize(n_dofs);
148 DoFToolsExt::map_dofs_to_support_points(MappingQ<dim, spacedim>(1),
149 dof_handler,
150 support_points);
151
152 // Generate a list of DoF indices starting from zero.
153 dof_indices.resize(n_dofs);
154 gen_linear_indices<vector_uta, types::global_dof_index>(dof_indices);
155
156 // Calculate the average mesh cell size at each support point.
157 dof_average_cell_size.assign(n_dofs, 0);
158 DoFToolsExt::map_dofs_to_average_cell_size(dof_handler,
159 dof_average_cell_size);
160}

183 dof_indices.resize(n_dofs);

185
186 // Calculate the average mesh cell size at each support point.
187 dof_average_cell_size.assign(n_dofs, 0);
188 DoFToolsExt::map_dofs_to_average_cell_size(dof_handler,
189 local_to_full_dof_id_map,

194template <int spacedim, typename Number>
195std::unique_ptr<ClusterTree<spacedim, Number>>
196ClusterTreeBuilder<spacedim, Number>::build() const
197{
198 // Create a cluster tree for all the DoF indices.
199 auto cluster_tree = std::make_unique<ClusterTree<spacedim, Number>>(
200 dof_indices, support_points, dof_average_cell_size, n_min);
201 // Partition the cluster tree.
202 cluster_tree->partition(support_points, dof_average_cell_size);
203
204 return cluster_tree;
205}
