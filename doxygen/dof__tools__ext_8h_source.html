69 template <int dim, int spacedim, typename SearchableMaterialIdContainer>
70 types::global_dof_index
71 extract_material_subdomain_dofs(
72 const DoFHandler<dim, spacedim> &dof_handler,
73 const SearchableMaterialIdContainer &material_ids,
74 std::vector<bool> &selected_dofs,
75 const bool reset_selectors_to_false = true)
76 {
77 AssertDimension(selected_dofs.size(), dof_handler.n_dofs());
78 types::global_dof_index effective_n_dofs = 0;
79
80 if (reset_selectors_to_false)
81 {
82 std::fill_n(selected_dofs.begin(), dof_handler.n_dofs(), false);
83 }
84 else
85 {
86 // Count the number of already selected DoFs.
87 for (auto selection_flag : selected_dofs)
88 if (selection_flag)
89 effective_n_dofs++;
90 }
91
92 // Global DoF indices for the current cell.
93 std::vector<types::global_dof_index> cell_dof_indices;
94 cell_dof_indices.reserve(
95 dof_handler.get_fe_collection().max_dofs_per_cell());
96
97 for (const auto &cell : dof_handler.active_cell_iterators())
98 {
99 // Find the current cell's material id in the given list.
100 auto found_iter = material_ids.find(cell->material_id());
101
102 if (found_iter != material_ids.end())
103 {
104 const unsigned int dofs_per_cell = cell->get_fe().dofs_per_cell;
105 cell_dof_indices.resize(dofs_per_cell);
106 cell->get_dof_indices(cell_dof_indices);
107 for (unsigned int i = 0; i < dofs_per_cell; ++i)
108 {
109 if (!selected_dofs[cell_dof_indices[i]])
110 {
111 selected_dofs[cell_dof_indices[i]] = true;
112 effective_n_dofs++;
113 }
114 }
115 }
116 }
117
118 return effective_n_dofs;
119 }

605 template <int dim, int spacedim, typename Number = double>
606 void
607 map_dofs_to_support_points_and_normal_vectors(
608 MappingInfo<dim, spacedim> &mapping_info,
609 const DoFHandler<dim, spacedim> &dof_handler,
610 std::vector<Point<spacedim, Number>> &support_points,
611 std::vector<Tensor<1, spacedim, Number>> &normal_vectors,
612 const bool is_normal_vector_negated = false)
613 {
614 const types::global_dof_index n_dofs = dof_handler.n_dofs();
615 const unsigned n_dofs_per_cell = dof_handler.get_fe().dofs_per_cell;
616 AssertDimension(n_dofs, support_points.size());
617 AssertDimension(n_dofs, normal_vectors.size());
618
619 // Flags indicating if a support point or normal vector at a DoF support
620 // point has been computed.
621 std::vector<bool> dof_flags;
622 dof_flags.assign(n_dofs, false);
623
624 // Get the unit support points of the finite element space.
625 const std::vector<Point<dim>> &fe_unit_support_points =
626 dof_handler.get_fe().get_unit_support_points();
627
628 std::unique_ptr<typename MappingQ<dim, spacedim>::InternalData>
629 &mapping_data = mapping_info.get_data();
630 const unsigned int mapping_n_shape_functions =
631 mapping_data->n_shape_functions;
632 mapping_info.resize_internal_data(n_dofs_per_cell);
633
634 // Compute mapping shape function values and their derivatives at the
635 // support points of the finite element.
636 mapping_data->compute_shape_function_values(fe_unit_support_points);
637
638 // Collect gradient matrices of mapping shape functions, which are evaluated
639 // at the finite element support points. N.B. The underlying data type for
640 // the matrix entries is double, which is consistent with the number type
641 // used by @p mapping_support_points below.
642 std::vector<LAPACKFullMatrixExt<double>> mapping_grad_matrices(
643 n_dofs_per_cell);
644 for (unsigned int d = 0; d < n_dofs_per_cell; d++)
645 {
646 mapping_grad_matrices[d].reinit(mapping_n_shape_functions, dim);
647 BEMTools::mappingq_shape_grad_matrix<dim, spacedim, double>(
648 *mapping_data, d, mapping_grad_matrices[d]);
649 }
650
651 // Iterate over each cell in the DoF handler.
652 std::vector<types::global_dof_index> cell_dof_indices(n_dofs_per_cell);
653 for (const auto &cell : dof_handler.active_cell_iterators())
654 {
655 // Compute the support points of the mapping in the current cell.
656 mapping_info.get_mapping().compute_mapping_support_points(cell);
657 const std::vector<Point<spacedim>> &mapping_support_points =
658 mapping_info.get_mapping().get_support_points();
659 // Get the DoF indices in the current cell.
660 cell->get_dof_indices(cell_dof_indices);
661
662 for (unsigned int d = 0; d < n_dofs_per_cell; d++)
663 {
664 const types::global_dof_index dof_index = cell_dof_indices[d];
665 if (!dof_flags[dof_index])
666 {
667 // Compute support point.
668 const Point<spacedim> fe_support_point =
669 mapping_info.get_mapping().transform_unit_to_real_cell(
670 fe_unit_support_points[d]);
671 support_points[dof_index] = fe_support_point;
672
673 // Compute normal vector.
674 Tensor<1, spacedim> normal_vector;
675 BEMTools::surface_jacobian_det_and_normal_vector(
676 mapping_support_points,
677 mapping_grad_matrices[d],
678 normal_vector,
679 is_normal_vector_negated);
680 // N.B. The locally computed @p normal_vector has the number
681 // type double, which may be different from the template
682 // argument Number.
683 normal_vectors[dof_index] = normal_vector;
684
685 dof_flags[dof_index] = true;
686 }
687 }
688 }
689 }
